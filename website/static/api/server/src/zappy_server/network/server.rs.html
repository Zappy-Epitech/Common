12/// Token used to identify the server listener in the event loop.
13const SERVER_TOKEN: Token = Token(0);
14
15/// The core server structure managing network connections and the event loop.
16pub struct Server {
17    /// Configuration parameters for the server.
18    config: ServerConfig,
19    /// The Mio poll instance for monitoring network events.
20    poll: Poll,
21    /// A map of connected clients, indexed by their unique Token.
22    clients: HashMap<Token, Client>,
23    /// Counter used to generate the next unique Token for a new client.
24    next_token: usize,
25    /// The game world of Trantor.
26    world: World,
27}
28
29impl Server {
30    /// Creates a new Server instance based on the provided configuration.
31    pub fn new(config: ServerConfig) -> io::Result<Self> {
32        let poll = Poll::new()?;
33        let world = World::new(
34            config.width,
35            config.height,
36            config.teams.clone(),
37            config.clients_nb,
38        );
39
40        Ok(Self {
41            config,
42            poll,
43            clients: HashMap::new(),
44            next_token: 1,
45            world,
46        })
47    }
48
49    /// Starts the main event loop of the server.
50    /// 
51    /// This method will block and handle new connections, incoming data,
52    /// and outgoing data until an error occurs.
53    pub fn run(&mut self) -> io::Result<()> {
54        let addr: SocketAddr = format!("0.0.0.0:{}", self.config.port).parse().unwrap();
55        let mut listener = TcpListener::bind(addr)?;
56
57        self.poll.registry().register(&mut listener, SERVER_TOKEN, Interest::READABLE)?;
58
59        let mut events = Events::with_capacity(128);
60
61        loop {
62            self.poll.poll(&mut events, Some(Duration::from_millis(100)))?;
63
64            for event in events.iter() {
65                match event.token() {
66                    SERVER_TOKEN => {
67                        while let Ok((mut stream, _)) = listener.accept() {
68                            let token = Token(self.next_token);
69                            self.next_token += 1;
70
71                            self.poll.registry().register(
72                                &mut stream,
73                                token,
74                                Interest::READABLE | Interest::WRITABLE,
75                            )?;
76
77                            let mut client = Client::new(stream);
78                            client.buffer_out.extend_from_slice(b"WELCOME\n");
79                            self.clients.insert(token, client);
80                        }
81                    }
82                    token => {
83                        if event.is_readable() {
84                            self.handle_read(token);
85                        }
86                        if event.is_writable() {
87                            self.handle_write(token);
88                        }
89                    }
90                }
91            }
92        }
93    }
94
95    /// Reads data from a client's socket into its input buffer.
96    fn handle_read(&mut self, token: Token) {
97        let mut closed = false;
98        let mut player_to_remove = None;
99
100        if let Some(client) = self.clients.get_mut(&token) {
101            let mut buf = [0; 1024];
102            loop {
103                match client.stream.read(&mut buf) {
104                    Ok(0) => {
105                        closed = true;
106                        if let ClientState::InGame(id) = client.state {
107                            player_to_remove = Some(id);
108                        }
109                        break;
110                    }
111                    Ok(n) => client.buffer_in.extend_from_slice(&buf[..n]),
112                    Err(ref e) if e.kind() == io::ErrorKind::WouldBlock => break,
113                    Err(_) => {
114                        closed = true;
115                        if let ClientState::InGame(id) = client.state {
116                            player_to_remove = Some(id);
117                        }
118                        break;
119                    }
120                }
121            }
122        }
123
124        if closed {
125            self.clients.remove(&token);
126            if let Some(id) = player_to_remove {
127                self.world.remove_player(id);
128            }
129            return;
130        }
131
132        self.process_buffer(token);
133    }
134
135    /// Processes the input buffer of a client, extracting complete lines (commands).
136    fn process_buffer(&mut self, token: Token) {
137        if let Some(client) = self.clients.get_mut(&token) {
138            while let Some(pos) = client.buffer_in.iter().position(|&b| b == b'\n') {
139                let line = client.buffer_in.drain(..pos + 1).collect::<Vec<u8>>();
140                let line_str = String::from_utf8_lossy(&line).trim().to_string();
141
142                match client.state {
143                    ClientState::Authenticating => {
144                        if line_str == "GRAPHIC" {
145                            client.state = ClientState::Graphic;
146                        } else if let Some(player_id) = self.world.add_player(&line_str) {
147                            client.state = ClientState::InGame(player_id);
148                            client.team_name = Some(line_str.clone());
149                            
150                            let slots = self.world.team_slots.get(&line_str).unwrap_or(&0);
151                            let msg = format!("{}\n{} {}\n", slots, self.config.width, self.config.height);
152                            client.buffer_out.extend_from_slice(msg.as_bytes());
153                        } else {
154                            client.buffer_out.extend_from_slice(b"ko\n");
155                        }
156                    }
157                    _ => {
158                    }
159                }
160            }
161        }
162    }
163
164    /// Writes data from a client's output buffer to its socket.
165    fn handle_write(&mut self, token: Token) {
166        if let Some(client) = self.clients.get_mut(&token) {
167            if !client.buffer_out.is_empty() {
168                match client.stream.write(&client.buffer_out) {
169                    Ok(n) => {
170                        client.buffer_out.drain(..n);
171                    }
172                    Err(ref e) if e.kind() == io::ErrorKind::WouldBlock => {}
173                    Err(_) => {
174                        self.clients.remove(&token);
175                    }
176                }
177            }
178        }
179    }
