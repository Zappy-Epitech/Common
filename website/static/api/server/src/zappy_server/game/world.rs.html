5#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
6pub enum Resource {
7    Food,
8    Linemate,
9    Deraumere,
10    Sibur,
11    Mendiane,
12    Phiras,
13    Thystame,
14}
15
16impl Resource {
17    pub fn all() -> &'static [Resource] {
18        &[
19            Resource::Food,
20            Resource::Linemate,
21            Resource::Deraumere,
22            Resource::Sibur,
23            Resource::Mendiane,
24            Resource::Phiras,
25            Resource::Thystame,
26        ]
27    }
28
29    pub fn density(self) -> f64 {
30        match self {
31            Resource::Food => 0.5,
32            Resource::Linemate => 0.3,
33            Resource::Deraumere => 0.15,
34            Resource::Sibur => 0.1,
35            Resource::Mendiane => 0.1,
36            Resource::Phiras => 0.08,
37            Resource::Thystame => 0.05,
38        }
39    }
40}

55pub struct World {
56    pub width: u32,
57    pub height: u32,
58    pub tiles: Vec<Tile>,
59    pub players: HashMap<usize, Player>,
60    pub team_slots: HashMap<String, usize>,
61    next_player_id: usize,
62}
63
64impl World {
65    pub fn new(width: u32, height: u32, teams: Vec<String>, clients_per_team: usize) -> Self {
66        let mut team_slots = HashMap::new();
67        for team in teams {
68            team_slots.insert(team, clients_per_team);
69        }
70
71        let mut world = Self {
72            width,
73            height,
74            tiles: vec![Tile::new(); (width * height) as usize],
75            players: HashMap::new(),
76            team_slots,
77            next_player_id: 1,
78        };
79        world.spawn_resources();
80        world
81    }
82
83    pub fn add_player(&mut self, team_name: &str, freq: u32) -> Option<usize> {
84        let slots = self.team_slots.get_mut(team_name)?;
85        if *slots == 0 {
86            return None;
87        }
88        *slots -= 1;
89
90        let mut rng = rand::thread_rng();
91        let id = self.next_player_id;
92        self.next_player_id += 1;
93
94        let x = rng.gen_range(0..self.width);
95        let y = rng.gen_range(0..self.height);
96        let direction = match rng.gen_range(0..4) {
97            0 => Direction::North,
98            1 => Direction::East,
99            2 => Direction::South,
100            _ => Direction::West,
101        };
102
103        let player = Player::new(id, x, y, direction, team_name.to_string(), freq);
104        self.players.insert(id, player);
105        Some(id)
106    }
107
108    pub fn remove_player(&mut self, player_id: usize) {
109        if let Some(player) = self.players.remove(&player_id) {
110            if let Some(slots) = self.team_slots.get_mut(&player.team) {
111                *slots += 1;
112            }
113        }
114    }
115
116    pub fn get_tile(&self, x: u32, y: u32) -> &Tile {
117        let idx = (y * self.width + x) as usize;
118        &self.tiles[idx]
119    }
120
121    pub fn get_tile_mut(&mut self, x: u32, y: u32) -> &mut Tile {
122        let idx = (y * self.width + x) as usize;
123        &mut self.tiles[idx]
124    }
125
126    pub fn spawn_resources(&mut self) {
127        let mut rng = rand::thread_rng();
128        let total_tiles = (self.width * self.height) as f64;
129
130        for &resource in Resource::all() {
131            let total_quantity = (total_tiles * resource.density()).ceil() as u32;
132            for _ in 0..total_quantity {
133                let x = rng.gen_range(0..self.width);
134                let y = rng.gen_range(0..self.height);
135                let tile = self.get_tile_mut(x, y);
136                *tile.resources.entry(resource).or_insert(0) += 1;
137            }
138        }
139    }
140}

146    #[test]
147    fn test_world_creation() {
148        let teams = vec!["Team1".to_string(), "Team2".to_string()];
149        let world = World::new(10, 10, teams, 5);
150
151        assert_eq!(world.width, 10);
152        assert_eq!(world.height, 10);
153        assert_eq!(world.tiles.len(), 100);
154        assert_eq!(world.team_slots.get("Team1"), Some(&5));
155    }
156
157    #[test]
158    fn test_resource_spawning() {
159        let world = World::new(10, 10, vec!["Team1".to_string()], 2);
160        
161        let mut total_food = 0;
162        for tile in &world.tiles {
163            total_food += tile.resources.get(&Resource::Food).unwrap_or(&0);
164        }
165        
166        assert!(total_food >= 50); // 100 * 0.5
167    }
168
169    #[test]
170    fn test_add_remove_player() {
171        let mut world = World::new(10, 10, vec!["Team1".to_string()], 1);
172        
173        let id = world.add_player("Team1", 100).expect("Should add player");
174        assert_eq!(world.team_slots.get("Team1"), Some(&0));
175        assert!(world.add_player("Team1", 100).is_none());
176
177        world.remove_player(id);
178        assert_eq!(world.team_slots.get("Team1"), Some(&1));
179    }
