1//! AI command dispatch and shared geometry.
2//!
3//! Two-phase command handling: [`init`] runs the immediate part of a command the
4//! moment it is received (e.g. starting an incantation), while [`execute`] runs
5//! the deferred effect once the command's time delay has elapsed, routing to the
6//! [`movement`], [`interaction`], [`social`] and [`incantation`] submodules. Also
7//! provides [`compute_direction`], the toroidal minimum-image direction used by
8//! `Broadcast` and `Eject`.
9

11pub mod interaction;
12pub mod social;
13pub mod incantation;

19/// Handles the immediate part of a command (RFC requirement).
20/// Only returns a message if the command requires an immediate response.
21pub fn init(command: &Command, player_id: usize, world: &mut World) -> Option<String> {
22    match command {
23        Command::Incantation => incantation::handle_start(player_id, world),
24        Command::Fork => {
25            world.gui_events.push_back(format!("pfk #{}\n", player_id));
26            None
27        }
28        _ => None,
29    }
30}

32/// Handles the final execution of a command after its time delay.
33pub fn execute(command: Command, player_id: usize, world: &mut World) -> String {
34    match command {
35        Command::Forward | Command::Right | Command::Left | Command::Eject => {
36            movement::execute(command, player_id, world)
37        }
38        Command::Look | Command::Inventory | Command::Take(_) | Command::Set(_) | Command::Fork => {
39            interaction::execute(command, player_id, world)
40        }
41        Command::Broadcast(_) => {
42            social::execute(command, player_id, world)
43        }
44        Command::Incantation => {
45            incantation::execute(player_id, world)
46        }
47        Command::ConnectNbr => {
48            if let Some(player) = world.players.get(&player_id) {
49                let team_name = player.team.clone();
50                let initial_slots = *world.team_slots.get(&team_name).unwrap_or(&0);
51                let egg_slots = world.eggs.iter().filter(|e| e.team == team_name).count();
52                format!("{}\n", initial_slots + egg_slots)
53            } else {
54                "0\n".to_string()
55            }
56        }
57    }
58}

60/// Calculates the relative direction (1-8) of a sound source on a toroidal map.
61/// 
62/// This implementation uses the **Minimum Image Convention** algorithm.
63/// 
64/// ### The Algorithm: Minimum Image Convention
65/// In a periodic (toroidal) world, a sound source has an infinite number of "images" 
66/// due to the map wrapping around. This algorithm ensures we always calculate the 
67/// path to the *closest* image of the sender.
68/// 
69/// 1. **Shortest Vector**: We calculate the raw distance `dx` and `dy`. If a distance 
70///    is greater than half the world size, we "wrap" it by subtracting/adding the 
71///    full world size. This gives us the shortest possible vector `(dx, dy)` on a torus.
72/// 
73/// 2. **Trigonometry**: We use `atan2(dx, -dy)` to convert this vector into a 
74///    geographic angle where 0° is North (upward in our grid).
75/// 
76/// 3. **Compass Mapping**: The angle is mapped to the RFC's 1-8 compass (8 slices of 45°).
77/// 
78/// 4. **Receiver Relativity**: Finally, we adjust the absolute direction by the 
79///    receiver's current orientation (`rdir`) so that '1' always represents the 
80///    tile directly in front of them.
81pub fn compute_direction(rx: i32, ry: i32, rdir: Direction, sx: i32, sy: i32, w: i32, h: i32) -> u32 {
82    if rx == sx && ry == sy { return 0; }
83
84    let mut dx = sx - rx;
85    let mut dy = sy - ry;
86
87    if dx.abs() > w / 2 { dx -= dx.signum() * w; }
88    if dy.abs() > h / 2 { dy -= dy.signum() * h; }
89
90    let angle = (dx as f64).atan2(-dy as f64).to_degrees();
91    let normalized_angle = if angle < 0.0 { angle + 360.0 } else { angle };
92
93    let abs_k = (((normalized_angle + 22.5) % 360.0) / 45.0).floor() as u32 + 1;
94
95    let dir_offset = match rdir {
96        Direction::North => 0,
97        Direction::East => 2,
98        Direction::South => 4,
99        Direction::West => 6,
100    };
101
102    let relative_k = if abs_k == 0 { 0 } else {
103        let mut k = abs_k as i32 - dir_offset;
104        while k <= 0 { k += 8; }
105        k as u32
106    };
107
108    relative_k
109}
