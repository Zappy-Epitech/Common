5/// Executes social and communication commands (Broadcast).
6pub fn execute(command: Command, player_id: usize, world: &mut World) -> String {
7    let text = match command {
8        Command::Broadcast(t) => t,
9        _ => return "ko\n".to_string(),
10    };
11
12    let (sender_x, sender_y) = {
13        let sender = world.players.get(&player_id).expect("Sender exists");
14        (sender.x, sender.y)
15    };
16
17    let mut targets = Vec::new();
18    for (&id, player) in world.players.iter() {
19        if id != player_id {
20            targets.push((id, player.x, player.y, player.direction));
21        }
22    }
23
24    for (target_id, target_x, target_y, target_dir) in targets {
25        let k = compute_direction(
26            target_x as i32, target_y as i32, target_dir,
27            sender_x as i32, sender_y as i32,
28            world.width as i32, world.height as i32
29        );
30        
31        let msg = format!("message {}, {}\n", k, text);
32        if let Some(player) = world.players.get_mut(&target_id) {
33            player.notifications.push_back(msg);
34        }
35    }
36
37    "ok\n".to_string()
38}

40/// Calculates the relative direction (1-8) of a sound source on a toroidal map.
41/// 
42/// This implementation uses the **Minimum Image Convention** algorithm.
43/// 
44/// ### The Algorithm: Minimum Image Convention
45/// In a periodic (toroidal) world, a sound source has an infinite number of "images" 
46/// due to the map wrapping around. This algorithm ensures we always calculate the 
47/// path to the *closest* image of the sender.
48/// 
49/// 1. **Shortest Vector**: We calculate the raw distance `dx` and `dy`. If a distance 
50///    is greater than half the world size, we "wrap" it by subtracting/adding the 
51///    full world size. This gives us the shortest possible vector `(dx, dy)` on a torus.
52/// 
53/// 2. **Trigonometry**: We use `atan2(dx, -dy)` to convert this vector into a 
54///    geographic angle where 0° is North (upward in our grid).
55/// 
56/// 3. **Compass Mapping**: The angle is mapped to the RFC's 1-8 compass (8 slices of 45°).
57/// 
58/// 4. **Receiver Relativity**: Finally, we adjust the absolute direction by the 
59///    receiver's current orientation (`rdir`) so that '1' always represents the 
60///    tile directly in front of them.
61fn compute_direction(rx: i32, ry: i32, rdir: Direction, sx: i32, sy: i32, w: i32, h: i32) -> u32 {
62    if rx == sx && ry == sy { return 0; }
63
64    let mut dx = sx - rx;
65    let mut dy = sy - ry;
66
67    if dx.abs() > w / 2 { dx -= dx.signum() * w; }
68    if dy.abs() > h / 2 { dy -= dy.signum() * h; }
69
70    let angle = (dx as f64).atan2(-dy as f64).to_degrees();
71    let normalized_angle = if angle < 0.0 { angle + 360.0 } else { angle };
72
73    let abs_k = (((normalized_angle + 22.5) % 360.0) / 45.0).floor() as u32 + 1;
74
75    let dir_offset = match rdir {
76        Direction::North => 0,
77        Direction::East => 2,
78        Direction::South => 4,
79        Direction::West => 6,
80    };
81
82    let relative_k = if abs_k == 0 { 0 } else {
83        let mut k = abs_k as i32 - dir_offset;
84        while k <= 0 { k += 8; }
85        k as u32
86    };
87
88    relative_k
89}
