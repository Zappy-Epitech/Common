1use std::collections::{HashMap, VecDeque};

6#[derive(Debug, Clone, Copy, PartialEq, Eq)]
7pub enum Direction {
8    North,
9    East,
10    South,
11    West,
12}
13
14impl Direction {
15    pub fn turn_right(self) -> Self {
16        match self {
17            Direction::North => Direction::East,
18            Direction::East => Direction::South,
19            Direction::South => Direction::West,
20            Direction::West => Direction::North,
21        }
22    }
23
24    pub fn turn_left(self) -> Self {
25        match self {
26            Direction::North => Direction::West,
27            Direction::West => Direction::South,
28            Direction::South => Direction::East,
29            Direction::East => Direction::North,
30        }
31    }
32}

34pub struct Player {
35    /// Unique identifier for the player.
36    pub id: usize,
37    /// Current X position on the map.
38    pub x: u32,
39    /// Current Y position on the map.
40    pub y: u32,
41    /// Current direction the player is facing.
42    pub direction: Direction,
43    /// Current elevation level of the player (starts at 1).
44    pub level: u32,
45    /// Player's inventory of resources.
46    pub inventory: HashMap<Resource, u32>,
47    /// The name of the team the player belongs to.
48    pub team: String,
49    /// Queue of commands waiting to be executed (max 10).
50    pub commands: VecDeque<PendingCommand>,
51    /// Time when the last queued command will finish.
52    pub last_command_end: Instant,
53    /// Precise moment of death due to hunger.
54    pub death_time: Instant,
55}
56
57impl Player {
58    /// Creates a new player at the given coordinates for a specific team.
59    /// 
60    /// Initial inventory contains 10 units of food.
61    /// `freq` is used to calculate the initial `death_time`.
62    pub fn new(id: usize, x: u32, y: u32, direction: Direction, team: String, freq: u32) -> Self {
63        let mut inventory = HashMap::new();
64        inventory.insert(Resource::Food, 10);
65
66        let now = Instant::now();
67        let life_duration = Duration::from_secs_f64(1260.0 / freq as f64);
68
69        Self {
70            id,
71            x,
72            y,
73            direction,
74            level: 1,
75            inventory,
76            team,
77            commands: VecDeque::new(),
78            last_command_end: now,
79            death_time: now + life_duration,
80        }
81    }
82}

101    #[test]
102    fn test_direction_turns() {
103        let mut dir = Direction::North;
104        
105        dir = dir.turn_right();
106        assert_eq!(dir, Direction::East);
107        
108        dir = dir.turn_right();
109        assert_eq!(dir, Direction::South);
110        
111        dir = dir.turn_left();
112        assert_eq!(dir, Direction::East);
113        
114        dir = dir.turn_left();
115        assert_eq!(dir, Direction::North);
116    }

118    #[test]
119    fn test_command_scheduling() {
120        let mut player = Player::new(1, 0, 0, Direction::North, "Team".to_string(), 100);
121        let now = Instant::now();
122        player.last_command_end = now;
123
124        let duration = Duration::from_secs_f64(7.0 / 100.0);
125        let end_time = now + duration;
126        player.commands.push_back(PendingCommand {
127            command: Command::Forward,
128            end_time,
129        });
130        player.last_command_end = end_time;
131
132        assert_eq!(player.commands.len(), 1);
133        assert_eq!(player.last_command_end, end_time);
134    }
