5pub fn execute(command: GuiCommand, world: &World, config: &ServerConfig) -> Vec<String> {
6    let mut responses = Vec::new();
7
8    match command {
9        GuiCommand::MapSize => {
10            responses.push(format!("msz {} {}\n", config.width, config.height));
11        }
12        GuiCommand::TileContent(x, y) => {
13            if x < config.width && y < config.height {
14                responses.push(format_bct(world, x, y));
15            } else {
16                responses.push("sbp\n".to_string());
17            }
18        }
19        GuiCommand::MapContent => {
20            for y in 0..config.height {
21                for x in 0..config.width {
22                    responses.push(format_bct(world, x, y));
23                }
24            }
25        }
26        GuiCommand::TeamNames => {
27            for team in &config.teams {
28                responses.push(format!("tna {}\n", team));
29            }
30        }
31        _ => {}
32    }
33
34    responses
35}

37pub fn format_bct(world: &World, x: u32, y: u32) -> String {
38    let tile = world.get_tile(x, y);
39    let f = tile.resources.get(&Resource::Food).unwrap_or(&0);
40    let l = tile.resources.get(&Resource::Linemate).unwrap_or(&0);
41    let d = tile.resources.get(&Resource::Deraumere).unwrap_or(&0);
42    let s = tile.resources.get(&Resource::Sibur).unwrap_or(&0);
43    let m = tile.resources.get(&Resource::Mendiane).unwrap_or(&0);
44    let p = tile.resources.get(&Resource::Phiras).unwrap_or(&0);
45    let t = tile.resources.get(&Resource::Thystame).unwrap_or(&0);
46    format!("bct {} {} {} {} {} {} {} {} {}\n", x, y, f, l, d, s, m, p, t)
47}
