1//! Time-related GUI commands: `sgt`, `sst`.
2//!
3//! Reports (`sgt`) and updates (`sst`) the server's time-unit frequency. Because
4//! all timers are stored as absolute [`Instant`]s, changing the frequency
5//! rescales every in-flight deadline (player death, queued commands and the next
6//! resource respawn) by the old/new frequency ratio so the perceived remaining
7//! game time is preserved.
8
9use crate::game::world::World;
10use crate::config::ServerConfig;
11use crate::protocol::gui::GuiCommand;
12use std::time::{Duration, Instant};
13
14/// Executes time-related GUI commands (`sgt`, `sst`).
15/// 
16/// Dynamically scales the remaining time of all ongoing events (death, commands, respawns)
17/// when the server frequency is modified.
18pub fn execute(command: GuiCommand, world: &mut World, config: &mut ServerConfig) -> Vec<String> {
19    match command {
20        GuiCommand::TimeRequest => {
21            vec![format!("sgt {}\n", config.freq)]
22        }
23        GuiCommand::TimeUpdate(new_freq) => {
24            if new_freq > 0 && new_freq != config.freq {
25                let old_freq = config.freq as f64;
26                let new_freq_f64 = new_freq as f64;
27                let ratio = old_freq / new_freq_f64;
28                let now = Instant::now();
29
30                for player in world.players.values_mut() {
31                    if player.death_time > now {
32                        let remaining = player.death_time.duration_since(now).as_secs_f64();
33                        player.death_time = now + Duration::from_secs_f64(remaining * ratio);
34                    }
35                    if player.last_command_end > now {
36                        let remaining = player.last_command_end.duration_since(now).as_secs_f64();
37                        player.last_command_end = now + Duration::from_secs_f64(remaining * ratio);
38                    }
39                    for cmd in player.commands.iter_mut() {
40                        if cmd.end_time > now {
41                            let remaining = cmd.end_time.duration_since(now).as_secs_f64();
42                            cmd.end_time = now + Duration::from_secs_f64(remaining * ratio);
43                        }
44                    }
45                }
46
47                if world.next_spawn_time > now {
48                    let remaining = world.next_spawn_time.duration_since(now).as_secs_f64();
49                    world.next_spawn_time = now + Duration::from_secs_f64(remaining * ratio);
50                }
51
52                world.freq = new_freq;
53                config.freq = new_freq;
54            }
55            vec![format!("sst {}\n", config.freq)]
56        }
57        _ => Vec::new(),
58    }
59}

61#[cfg(test)]
62mod tests {
63    use super::*;
64
65    #[test]
66    fn test_time_commands() {
67        let mut config = ServerConfig {
68            port: 4242,
69            width: 10,
70            height: 10,
71            teams: vec!["TeamA".to_string()],
72            clients_nb: 5,
73            freq: 100,
74            bonus: false,
75        };
76        let mut world = World::new(config.width, config.height, config.teams.clone(), config.clients_nb, config.freq);
77
78        let res_sgt = execute(GuiCommand::TimeRequest, &mut world, &mut config);
79        assert_eq!(res_sgt.len(), 1);
80        assert_eq!(res_sgt[0], "sgt 100\n");
81
82        let res_sst = execute(GuiCommand::TimeUpdate(200), &mut world, &mut config);
83        assert_eq!(res_sst.len(), 1);
84        assert_eq!(res_sst[0], "sst 200\n");
85        assert_eq!(config.freq, 200);
86        assert_eq!(world.freq, 200);
87    }
