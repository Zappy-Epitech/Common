1//! Player-related GUI commands: `ppo`, `plv`, `pin`.
2//!
3//! Answers graphic-client queries about a specific player — position and
4//! orientation, elevation level, and inventory — replying `sbp` when the
5//! requested player id does not exist.
6
7use crate::game::world::{World, Resource};
8use crate::game::player::Direction;
9use crate::protocol::gui::GuiCommand;
10
11/// Executes player-related GUI commands (`ppo`, `plv`, `pin`).
12pub fn execute(command: GuiCommand, world: &World) -> Vec<String> {
13    let mut responses = Vec::new();
14
15    match command {
16        GuiCommand::PlayerPosition(id) => {
17            if let Some(player) = world.players.get(&id) {
18                let orientation = match player.direction {
19                    Direction::North => 1,
20                    Direction::East => 2,
21                    Direction::South => 3,
22                    Direction::West => 4,
23                };
24                responses.push(format!("ppo #{} {} {} {}\n", id, player.x, player.y, orientation));
25            } else {
26                responses.push("sbp\n".to_string());
27            }
28        }
29        GuiCommand::PlayerLevel(id) => {
30            if let Some(player) = world.players.get(&id) {
31                responses.push(format!("plv #{} {}\n", id, player.level));
32            } else {
33                responses.push("sbp\n".to_string());
34            }
35        }
36        GuiCommand::PlayerInventory(id) => {
37            if let Some(player) = world.players.get(&id) {
38                let f = player.get_food_count(world.freq);
39                let l = player.inventory.get(&Resource::Linemate).unwrap_or(&0);
40                let d = player.inventory.get(&Resource::Deraumere).unwrap_or(&0);
41                let s = player.inventory.get(&Resource::Sibur).unwrap_or(&0);
42                let m = player.inventory.get(&Resource::Mendiane).unwrap_or(&0);
43                let p = player.inventory.get(&Resource::Phiras).unwrap_or(&0);
44                let t = player.inventory.get(&Resource::Thystame).unwrap_or(&0);
45                
46                responses.push(format!("pin #{} {} {} {} {} {} {} {} {} {}\n", 
47                    id, player.x, player.y, f, l, d, s, m, p, t));
48            } else {
49                responses.push("sbp\n".to_string());
50            }
51        }
52        _ => {}
53    }
54
55    responses
56}

58#[cfg(test)]
59mod tests {
60    use super::*;
61
62    #[test]
63    fn test_player_info_commands() {
64        let mut world = World::new(10, 10, vec!["TeamA".to_string()], 5, 100);
65        let player_id = world.add_player("TeamA", 100).expect("Could not add player in test");
66        
67        // ppo
68        let res_ppo = execute(GuiCommand::PlayerPosition(player_id), &world);
69        assert_eq!(res_ppo.len(), 1);
70        assert!(res_ppo[0].starts_with(&format!("ppo #{}", player_id)));
71
72        // plv
73        let res_plv = execute(GuiCommand::PlayerLevel(player_id), &world);
74        assert_eq!(res_plv.len(), 1);
75        assert_eq!(res_plv[0], format!("plv #{} 1\n", player_id));
76
77        // pin
78        let res_pin = execute(GuiCommand::PlayerInventory(player_id), &world);
79        assert_eq!(res_pin.len(), 1);
80        assert!(res_pin[0].starts_with(&format!("pin #{}", player_id)));
81
82        // Invalid player
83        let res_invalid = execute(GuiCommand::PlayerLevel(999), &world);
84        assert_eq!(res_invalid[0], "sbp\n");
85    }
